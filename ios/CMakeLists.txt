cmake_minimum_required(VERSION 3.24)
project(Kioku LANGUAGES C CXX OBJCXX)

find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Network Svg)

# Only the app goes into an archive (tools/testflight.sh); libraries with
# SKIP_INSTALL=NO would make it a generic archive that cannot be exported
set(CMAKE_XCODE_ATTRIBUTE_SKIP_INSTALL YES)

# ANKI_TAG, PYQT6_SIP_VERSION, ... from ../shared/versions.env
file(STRINGS ${CMAKE_SOURCE_DIR}/../shared/versions.env VERSION_LINES REGEX "^[A-Z_0-9]+=")
foreach(line IN LISTS VERSION_LINES)
    string(REGEX MATCH "^([A-Z_0-9]+)=(.*)$" _ "${line}")
    set(${CMAKE_MATCH_1} "${CMAKE_MATCH_2}")
endforeach()

set(KIOKU_VERSION "${ANKI_PY_VERSION}" CACHE STRING "App version shown in Settings")
set(KIOKU_BUILD "1" CACHE STRING "Build number (CFBundleVersion)")

set(DEPS ${CMAKE_SOURCE_DIR}/deps)
set(PYTHON_XCFRAMEWORK ${DEPS}/python-ios/Python.xcframework)
if(CMAKE_OSX_SYSROOT MATCHES "simulator")
    set(PYTHON_SLICE ${PYTHON_XCFRAMEWORK}/ios-arm64_x86_64-simulator)
    set(RUST_TARGET aarch64-apple-ios-sim)
    set(NATIVE_TARGET ios-sim)
else()
    set(PYTHON_SLICE ${PYTHON_XCFRAMEWORK}/ios-arm64)
    set(RUST_TARGET aarch64-apple-ios)
    set(NATIVE_TARGET ios)
endif()
set(KIOKU_PYTHON_INCLUDE ${PYTHON_SLICE}/include/python${PYTHON_VERSION})

# PyQt6 bindings (generated by sip-build, see tools/gen_pyqt.sh)
include(cmake/pyqt6.cmake)

file(GLOB SIP_SOURCES ${DEPS}/sdist/pyqt6_sip-${PYQT6_SIP_VERSION}/*.c)
add_library(pyqt6_sip STATIC ${SIP_SOURCES})
target_include_directories(pyqt6_sip PRIVATE ${KIOKU_PYTHON_INCLUDE})
target_compile_options(pyqt6_sip PRIVATE -w)

# Rust extension modules, built as static libraries by tools/build_rust.sh
set(RUST_LIBS
    ${DEPS}/anki/target/${RUST_TARGET}/release/librsbridge.a
    ${DEPS}/sdist/orjson-${ORJSON_VERSION}/target/${RUST_TARGET}/release/liborjson.a
    ${DEPS}/sdist/rpds_py-${RPDS_VERSION}/target/${RUST_TARGET}/release/librpds.a
)
# PyMuPDF's pymupdf._mupdf and pymupdf._extra (shared/tools/build_pymupdf.sh)
set(PYMUPDF_LIB ${CMAKE_SOURCE_DIR}/../build/pymupdf/${NATIVE_TARGET}/lib/libpymupdf.a)
# _kioku_programs: the programs add-ons start, run inside the app
# (shared/tools/build_programs.sh)
set(PROGRAMS_LIB ${CMAKE_SOURCE_DIR}/../build/programs/${NATIVE_TARGET}/lib/libkiokuprog.a)

# app icon (shared/tools/make_icon.py), compiled by Xcode's asset catalog tool
# and the privacy manifest App Store Connect requires (res/PrivacyInfo.xcprivacy)
set(KIOKU_ASSETS ${CMAKE_SOURCE_DIR}/res/Assets.xcassets)
set(KIOKU_PRIVACY ${CMAKE_SOURCE_DIR}/res/PrivacyInfo.xcprivacy)
set_source_files_properties(${KIOKU_ASSETS} ${KIOKU_PRIVACY}
    PROPERTIES MACOSX_PACKAGE_LOCATION Resources)

qt_add_executable(Kioku MACOSX_BUNDLE src/main.mm ${KIOKU_ASSETS} ${KIOKU_PRIVACY})
target_include_directories(Kioku PRIVATE ${KIOKU_PYTHON_INCLUDE})
target_link_libraries(Kioku PRIVATE
    pyqt6_QtCore pyqt6_QtGui pyqt6_QtWidgets pyqt6_QtNetwork pyqt6_QtSvg
    pyqt6_sip
    ${RUST_LIBS}
    ${PYMUPDF_LIB}
    ${PROGRAMS_LIB}
    Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Network Qt6::Svg
    "-F${PYTHON_SLICE}" "-framework Python"
    "-framework WebKit" "-framework JavaScriptCore" "-framework UIKit" "-framework Security"
    "-framework SystemConfiguration" "-framework CoreFoundation"
    "-framework AVFoundation"
    "-framework VideoToolbox" "-framework AudioToolbox" "-framework CoreMedia"
    "-framework CoreVideo" "-framework CoreServices"
    iconv z bz2 resolv
)

set_target_properties(Kioku PROPERTIES
    OUTPUT_NAME Kioku
    MACOSX_BUNDLE_INFO_PLIST ${CMAKE_SOURCE_DIR}/src/Info.plist.in
    MACOSX_BUNDLE_GUI_IDENTIFIER io.ankidesk.kioku
    MACOSX_BUNDLE_BUNDLE_NAME Kioku
    MACOSX_BUNDLE_BUNDLE_VERSION ${KIOKU_BUILD}
    MACOSX_BUNDLE_SHORT_VERSION_STRING ${KIOKU_VERSION}
    XCODE_ATTRIBUTE_PRODUCT_BUNDLE_IDENTIFIER io.ankidesk.kioku
    XCODE_ATTRIBUTE_IPHONEOS_DEPLOYMENT_TARGET 17.0
    XCODE_ATTRIBUTE_TARGETED_DEVICE_FAMILY "1,2"
    XCODE_ATTRIBUTE_ENABLE_BITCODE NO
    XCODE_ATTRIBUTE_DEAD_CODE_STRIPPING YES
    XCODE_ATTRIBUTE_ASSETCATALOG_COMPILER_APPICON_NAME AppIcon
    XCODE_ATTRIBUTE_LD_RUNPATH_SEARCH_PATHS "@executable_path/Frameworks"
    XCODE_ATTRIBUTE_DEVELOPMENT_TEAM "${KIOKU_TEAM}"
    XCODE_ATTRIBUTE_CODE_SIGN_STYLE Automatic
    # a scheme, so that tools/testflight.sh can archive
    XCODE_GENERATE_SCHEME ON
    XCODE_ATTRIBUTE_SKIP_INSTALL NO
    XCODE_ATTRIBUTE_INSTALL_PATH "$(LOCAL_APPS_DIR)"
    XCODE_EMBED_FRAMEWORKS ${PYTHON_SLICE}/Python.framework
    XCODE_EMBED_FRAMEWORKS_CODE_SIGN_ON_COPY ON
    XCODE_EMBED_FRAMEWORKS_REMOVE_HEADERS_ON_COPY ON
)

# Copy the Python standard library, Anki and its dependencies into the
# bundle, turning extension modules into signed frameworks.
add_custom_command(TARGET Kioku POST_BUILD
    COMMAND ${CMAKE_SOURCE_DIR}/tools/bundle_python.sh
            ${PYTHON_SLICE} ${CMAKE_SOURCE_DIR}
    VERBATIM)
