cmake_minimum_required(VERSION 3.24)
project(Kioku LANGUAGES C CXX)

find_package(Qt6 REQUIRED COMPONENTS Core CorePrivate Gui Widgets Network Svg)

# ANKI_TAG, PYQT6_SIP_VERSION, ... from ../shared/versions.env
file(STRINGS ${CMAKE_SOURCE_DIR}/../shared/versions.env VERSION_LINES REGEX "^[A-Z_0-9]+=")
foreach(line IN LISTS VERSION_LINES)
    string(REGEX MATCH "^([A-Z_0-9]+)=(.*)$" _ "${line}")
    set(${CMAKE_MATCH_1} "${CMAKE_MATCH_2}")
endforeach()

set(KIOKU_VERSION "${ANKI_PY_VERSION}" CACHE STRING "App version (versionName)")
set(KIOKU_BUILD "1" CACHE STRING "Build number (versionCode)")
# a copy of package/ with assets/python.zip, made by tools/build.sh
set(KIOKU_PACKAGE_DIR "${CMAKE_BINARY_DIR}/package" CACHE PATH "Android package sources")

set(DEPS ${CMAKE_SOURCE_DIR}/deps)
set(PYTHON_PREFIX ${DEPS}/python-android/prefix)
set(KIOKU_PYTHON_INCLUDE ${PYTHON_PREFIX}/include/python${PYTHON_VERSION})
set(RUST_TARGET aarch64-linux-android)

# PyQt6 bindings (generated by sip-build, see tools/gen_pyqt.sh)
include(cmake/pyqt6.cmake)

file(GLOB SIP_SOURCES ${DEPS}/sdist/pyqt6_sip-${PYQT6_SIP_VERSION}/*.c)
add_library(pyqt6_sip STATIC ${SIP_SOURCES})
target_include_directories(pyqt6_sip PRIVATE ${KIOKU_PYTHON_INCLUDE})
target_compile_options(pyqt6_sip PRIVATE -w)

# Rust extension modules, built as static libraries by tools/build_rust.sh
set(RUST_LIBS
    ${DEPS}/anki/target/${RUST_TARGET}/release/librsbridge.a
    ${DEPS}/sdist/orjson-${ORJSON_VERSION}/target/${RUST_TARGET}/release/liborjson.a
    ${DEPS}/sdist/rpds_py-${RPDS_VERSION}/target/${RUST_TARGET}/release/librpds.a
)

qt_add_executable(Kioku src/main.cpp src/kioku_android.cpp)
target_include_directories(Kioku PRIVATE ${KIOKU_PYTHON_INCLUDE})
target_compile_definitions(Kioku PRIVATE KIOKU_PYTHON_VERSION="${PYTHON_VERSION}"
    KIOKU_ANDROID_ABI="${ANDROID_ABI}")
target_link_libraries(Kioku PRIVATE
    pyqt6_QtCore pyqt6_QtGui pyqt6_QtWidgets pyqt6_QtNetwork pyqt6_QtSvg
    pyqt6_sip
    ${RUST_LIBS}
    Qt6::Core Qt6::CorePrivate Qt6::Gui Qt6::Widgets Qt6::Network Qt6::Svg
    ${PYTHON_PREFIX}/lib/libpython${PYTHON_VERSION}.so
    log android z
)
# Each Rust staticlib carries its own copy of std; their only unmangled
# common symbol, rust_eh_personality, behaves the same in every copy.
target_link_options(Kioku PRIVATE "-Wl,--allow-multiple-definition")

set_target_properties(Kioku PROPERTIES
    QT_ANDROID_PACKAGE_NAME io.ankidesk.kioku
    QT_ANDROID_APP_NAME Kioku
    QT_ANDROID_APP_ICON "@mipmap/ic_launcher"
    QT_ANDROID_PACKAGE_SOURCE_DIR ${KIOKU_PACKAGE_DIR}
    QT_ANDROID_MIN_SDK_VERSION 28
    QT_ANDROID_TARGET_SDK_VERSION 35
    QT_ANDROID_COMPILE_SDK_VERSION 36
    QT_ANDROID_VERSION_CODE ${KIOKU_BUILD}
    QT_ANDROID_VERSION_NAME ${KIOKU_VERSION}
    # libkiokuprog.so: the programs add-ons start (an executable, see
    # shared/tools/build_programs.sh)
    QT_ANDROID_EXTRA_LIBS "${PYTHON_PREFIX}/lib/libpython${PYTHON_VERSION}.so;${PYTHON_PREFIX}/lib/libssl_python.so;${PYTHON_PREFIX}/lib/libcrypto_python.so;${PYTHON_PREFIX}/lib/libsqlite3_python.so;${CMAKE_SOURCE_DIR}/../build/programs/android/lib/libkiokuprog.so"
)
